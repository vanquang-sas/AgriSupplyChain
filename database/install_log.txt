SQL> CREATE OR REPLACE TRIGGER TRG_CTDH_TONGTIEN
  2  AFTER INSERT OR UPDATE OR DELETE ON CHITIETDONHANG
  3  FOR EACH ROW
  4  DECLARE
  5  	 v_Delta NUMBER := 0;
  6  	 v_MaDH VARCHAR2(10);
  7  	 v_LoaiKH NVARCHAR2(50);
  8  
  9  	 v_TyLeGiamGia NUMBER := 0;
 10  	 v_TongTienHangMoi NUMBER := 0;
 11  	 v_PhiVanChuyen NUMBER := 0;
 12  	 v_GiamGiaMoi NUMBER := 0;
 13  BEGIN
 14  	 -- 1. Xác định mức thay đổi của ThanhTien (Delta)
 15  	 IF INSERTING THEN
 16  	     v_Delta := :NEW.ThanhTien;
 17  	     v_MaDH := :NEW.MaDH;
 18  	 ELSIF UPDATING THEN
 19  	     v_Delta := :NEW.ThanhTien - :OLD.ThanhTien;
 20  	     v_MaDH := :NEW.MaDH;
 21  	 ELSIF DELETING THEN
 22  	     v_Delta := -:OLD.ThanhTien;
 23  	     v_MaDH := :OLD.MaDH;
 24  	 END IF;
 25  
 26  	 -- 2. Lấy Tổng tiền hàng hiện tại, Phí vận chuyển và Loại khách hàng
 27  	 SELECT DH.TongTienHang, DH.PhiVanChuyen, KH.LoaiKH
 28  	 INTO v_TongTienHangMoi, v_PhiVanChuyen, v_LoaiKH
 29  	 FROM DONHANG DH
 30  	 JOIN KHACHHANG KH ON DH.MaKH = KH.MaKH
 31  	 WHERE DH.MaDH = v_MaDH;

 33  	 -- Cộng dồn Delta để ra Tổng tiền hàng mới
 34  	 v_TongTienHangMoi := NVL(v_TongTienHangMoi, 0) + v_Delta;

 37  	 -- Giả sử giá trị lưu trong THAMSO là số thập phân (Ví dụ: 0.05 tương đương 5%)

 51  	 END;

SQL> CREATE OR REPLACE PROCEDURE SP_XACNHAN_VITRI_CTLH (
  2  p_MaCTLH IN VARCHAR2,
  3  	 p_MaKho IN VARCHAR2,	    -- Truyền từ ComboBox Kho trên giao diện
  4  	 p_TGHetHan IN DATE,	    -- Truyền từ DatePicker trên giao diện
  5  	 p_ViTri IN NVARCHAR2	    -- Truyền từ ô Text Vị trí trên giao diện
  6  ) IS
  7  	 v_MaLH VARCHAR2(10);
  8  	 v_MaSP VARCHAR2(10);
  9  	 v_SoLuong NUMBER(10,2);
 10  	 v_BaoQuan NVARCHAR2(100);
 11  	 v_LoaiKho NVARCHAR2(100);
 12  	 v_TonTai NUMBER;
 13  	 v_ChuaXepXong NUMBER;
 14  	 v_TrangThaiLH NVARCHAR2(50);
 15  BEGIN
 16  	 -- 1. Lấy thông tin Chi tiết lô hàng và Yêu cầu bảo quản của sản phẩm
 17  	 SELECT CTLH.MaLH, CTLH.MaSP, CTLH.SoLuong, SP.BaoQuan
 18  	 INTO v_MaLH, v_MaSP, v_SoLuong, v_BaoQuan
 19  	 FROM CHITIETLOHANG CTLH
 20  	 JOIN SANPHAM SP ON CTLH.MaSP = SP.MaSP
 21  	 WHERE CTLH.MaCTLH = p_MaCTLH;
 22  
 23  	 -- Kiểm tra trạng thái Lô hàng (Chỉ xử lý khi đang chờ nhập kho)
 24  	 SELECT TrangThaiLH INTO v_TrangThaiLH FROM LOHANG WHERE MaLH = v_MaLH;
 25  	 IF v_TrangThaiLH != 'Chờ nhập kho' THEN
 26  	     RAISE_APPLICATION_ERROR(-20026, 'Lô hàng không ở trạng thái Chờ nhập kho!');
 27  	 END IF;
 28  
 29  	 -- 2. ĐỐI CHIẾU AN TOÀN: Kiểm tra Kho nhân viên chọn có khớp chuẩn bảo quản không
 30  	 SELECT LoaiKho INTO v_LoaiKho FROM KHO WHERE MaKho = p_MaKho;
 31  	 IF v_LoaiKho != v_BaoQuan THEN
 32  	     RAISE_APPLICATION_ERROR(-20028, 'Bảo quản sai quy cách! Sản phẩm ' || v_MaSP || ' yêu cầu kho [' || v_BaoQuan || '] nhưng bạn lại chọn kho [' || v_LoaiKho || '].');
 33  	 END IF;
 34  
 35  	 -- 3. Kiểm tra chống nhập đúp (1 mã chi tiết chỉ được cất 1 lần)
 36  	 SELECT COUNT(*) INTO v_TonTai FROM TONKHO WHERE MaCTLH = p_MaCTLH;
 37  	 IF v_TonTai > 0 THEN
 38  	     RAISE_APPLICATION_ERROR(-20027, 'Sản phẩm của chi tiết lô hàng này đã được xác nhận cất vào kho rồi!');
 39  	 END IF;
 40  
 41  	 -- 4. INSERT dữ liệu chính thức vào bảng Tồn Kho
 42  	 INSERT INTO TONKHO (MaKho, MaCTLH, SLConLai, SLKhaDung, TGNhapKho, TGHetHan, ViTri)
 43  	 VALUES (p_MaKho, p_MaCTLH, v_SoLuong, v_SoLuong, SYSDATE, p_TGHetHan, p_ViTri);
 44  
 45  	 -- 5. TỰ ĐỘNG ĐÓNG LÔ HÀNG: Đếm xem Lô hàng này còn sản phẩm nào chưa cất không?
 46  	 SELECT COUNT(*) INTO v_ChuaXepXong
 47  	 FROM CHITIETLOHANG C
 48  	 WHERE C.MaLH = v_MaLH
 49  	   AND C.MaCTLH NOT IN (SELECT MaCTLH FROM TONKHO);
 50  
 51  	 IF v_ChuaXepXong = 0 THEN
 52  	     -- Nếu tất cả đã cất xong -> Cập nhật Lô hàng hoàn tất
 53  	     UPDATE LOHANG SET TrangThaiLH = 'Đã nhập kho' WHERE MaLH = v_MaLH;
 54  	 END IF;
 55  
 56  	 -- Không COMMIT và EXCEPTION ROLLBACK ở đây.
 57  	 -- Vì NẾU COMMIT dữ liệu sẽ được lưu ngay vào DB và có thể dẫn đến bị trùng
 58  	 -- Vì hệ thống sẽ lưu nhiều dòng cùng một lúc (Batch), nếu xảy ra lỗi ở dòng số 3,
 59  	 -- hệ thống phải huỷ (rollback) toàn bộ các dòng 1 và 2 đã chạy trước đó để tránh tình trạng dữ liệu lưu nửa vời.

SQL> CREATE OR REPLACE PROCEDURE SP_YEUCAU_XUATKHO (p_MaDH IN VARCHAR2) IS
  2  	 v_SoLuongCan NUMBER;
  3  	 v_SoLuongXuat NUMBER;
  4  	 v_TrangThaiDH NVARCHAR2(50);
  5  BEGIN
  6  	 SELECT TrangThaiDH INTO v_TrangThaiDH FROM DONHANG WHERE MaDH = p_MaDH;
  7  
  8  	 -- Kiểm tra đơn hàng có đủ điều kiện để phân bổ hàng không
  9  	 IF v_TrangThaiDH IN ('Đang giao', 'Hoàn thành', 'Đã huỷ') THEN
 10  	     RAISE_APPLICATION_ERROR(-20023, 'Trạng thái đơn hàng không hợp lệ!');
 11  	 END IF;
 12  
 13  	 -- Duyệt từng sản phẩm trong chi tiết đơn hàng
 14  	 FOR rec_CTDH IN (SELECT MaCTDH, MaSP, SoLuong FROM CHITIETDONHANG WHERE MaDH = p_MaDH) LOOP
 15  	     v_SoLuongCan := rec_CTDH.SoLuong;
 16  
 17  	     -- Tìm hàng theo FEFO (Hết hạn trước xuất trước) và FIFO (Nhập trước xuất trước)
 18  	     FOR rec_TK IN (
 19  		 SELECT TK.MaTonKho, TK.SLKhaDung
 20  		 FROM TONKHO TK
 21  		 JOIN CHITIETLOHANG CTLH ON TK.MaCTLH = CTLH.MaCTLH
 22  		 WHERE CTLH.MaSP = rec_CTDH.MaSP
 23  		   AND TK.SLKhaDung > 0
 24  		   AND TK.TGHetHan >= TRUNC(SYSDATE)
 25  		 ORDER BY TK.TGHetHan ASC, TK.TGNhapKho ASC
 26  		 FOR UPDATE -- Thêm dòng này để lock record
 27  	     ) LOOP
 28  		 EXIT WHEN v_SoLuongCan = 0;
 29  
 30  		 IF rec_TK.SLKhaDung >= v_SoLuongCan THEN
 31  		     v_SoLuongXuat := v_SoLuongCan;
 32  		     v_SoLuongCan := 0;
 33  		 ELSE
 34  		     v_SoLuongXuat := rec_TK.SLKhaDung;
 35  		     v_SoLuongCan := v_SoLuongCan - rec_TK.SLKhaDung;
 36  		 END IF;
 37  
 38  		 -- Insert với MaNV để NULL và trạng thái 'Tạm giữ'
 39  		 -- Trigger sẽ tự động trừ SLKhaDung trong bảng TONKHO
 40  		 INSERT INTO XUATKHO (MaCTDH, MaTonKho, MaNV, SLXuat, TGCapNhat, TrangThaiXK)
 41  		 VALUES (rec_CTDH.MaCTDH, rec_TK.MaTonKho, NULL, v_SoLuongXuat, SYSDATE, 'Tạm giữ');
 42  
 43  	     END LOOP;
 44  
 45  	     IF v_SoLuongCan > 0 THEN
 46  		 RAISE_APPLICATION_ERROR(-20024, 'Kho không đủ hàng cho sản phẩm: ' || rec_CTDH.MaSP);
 47  	     END IF;
 48  	 END LOOP;
 49  
 50  	 COMMIT;
