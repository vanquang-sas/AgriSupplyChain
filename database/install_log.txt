SQL> CREATE OR REPLACE FUNCTION FN_TINH_PHIVANCHUYEN (
  2  	 p_MaKH IN VARCHAR2
  3  ) RETURN NUMBER
  4  IS
  5  	 v_LoaiKH NVARCHAR2(50) := 'Thường';
  6  	 v_PhiVanChuyen NUMBER := 0;
  7  BEGIN
  8  	 -- Lấy loại khách hàng

SQL> -- 5. Thống kê tài chính
SQL> CREATE OR REPLACE FUNCTION FN_THONGKE_TAICHINH (
  2  	 p_LoaiThongKe IN VARCHAR2, -- 'MONTH' hoặc 'DAY'
  3  	 p_Period IN NUMBER
  4  ) RETURN SYS_REFCURSOR
  5  AS
  6  	 v_cursor SYS_REFCURSOR;
  7  	 v_AnchorDate DATE := SYSDATE;
  8  BEGIN
  9  	 IF p_LoaiThongKe = 'MONTH' THEN
 10  	     OPEN v_cursor FOR
 11  	     WITH MonthRange AS (
 12  		 SELECT ADD_MONTHS(TRUNC(v_AnchorDate, 'MM'), -LEVEL + 1) AS ThoiGian
 13  		 FROM DUAL
 14  		 CONNECT BY LEVEL <= p_Period
 15  	     ),
 16  	     RevenueData AS (
 17  		 SELECT TRUNC(TGDat, 'MM') AS ThoiGian, SUM(TongTien) AS DoanhThu
 18  		 FROM DONHANG
 19  		 WHERE (TrangThaiDH = 'HoanThanh' OR TrangThaiDH = 'Hoàn thành')
 20  		   AND TGDat >= ADD_MONTHS(TRUNC(v_AnchorDate, 'MM'), -p_Period + 1)
 21  		 GROUP BY TRUNC(TGDat, 'MM')
 22  	     ),
 23  	     CostData AS (
 24  		 SELECT TRUNC(TGNhap, 'MM') AS ThoiGian, SUM(TongTien) AS ChiPhi
 25  		 FROM LOHANG
 26  		 WHERE TrangThaiLH = N'Đã nhập kho'
 27  		   AND TGNhap >= ADD_MONTHS(TRUNC(v_AnchorDate, 'MM'), -p_Period + 1)
 28  		 GROUP BY TRUNC(TGNhap, 'MM')
 29  	     )
 30  	     SELECT
 31  		 TO_CHAR(mr.ThoiGian, 'MM/YYYY') AS ThangNam,
 32  		 NVL(rd.DoanhThu, 0) AS DoanhThu,
 33  		 NVL(cd.ChiPhi, 0) AS ChiPhi
 34  	     FROM MonthRange mr
 35  	     LEFT JOIN RevenueData rd ON mr.ThoiGian = rd.ThoiGian
 36  	     LEFT JOIN CostData cd ON mr.ThoiGian = cd.ThoiGian
 37  	     ORDER BY mr.ThoiGian ASC;
 38  
 39  	 ELSE -- Xử lý cho chế độ theo NGÀY
 40  	     OPEN v_cursor FOR
 41  	     WITH DayRange AS (
 42  		 SELECT TRUNC(v_AnchorDate) - LEVEL + 1 AS ThoiGian
 43  		 FROM DUAL
 44  		 CONNECT BY LEVEL <= p_Period
 45  	     ),
 46  	     RevenueData AS (
 47  		 SELECT TRUNC(TGDat) AS ThoiGian, SUM(TongTien) AS DoanhThu
 48  		 FROM DONHANG
 49  		 WHERE (TrangThaiDH = 'HoanThanh' OR TrangThaiDH = 'Hoàn thành')
 50  		   AND TGDat >= TRUNC(v_AnchorDate) - p_Period + 1
 51  		 GROUP BY TRUNC(TGDat)
 52  	     ),
 53  	     CostData AS (
 54  		 SELECT TRUNC(TGNhap) AS ThoiGian, SUM(TongTien) AS ChiPhi
 55  		 FROM LOHANG
 56  		 WHERE TrangThaiLH = N'Đã nhập kho'
 57  		   AND TGNhap >= TRUNC(v_AnchorDate) - p_Period + 1
 58  		 GROUP BY TRUNC(TGNhap)
 59  	     )
 60  	     SELECT
 61  		 TO_CHAR(dr.ThoiGian, 'DD/MM/YYYY') AS ThangNam,
 62  		 NVL(rd.DoanhThu, 0) AS DoanhThu,
 63  		 NVL(cd.ChiPhi, 0) AS ChiPhi
 64  	     FROM DayRange dr
 65  	     LEFT JOIN RevenueData rd ON dr.ThoiGian = rd.ThoiGian
 66  	     LEFT JOIN CostData cd ON dr.ThoiGian = cd.ThoiGian
 67  	     ORDER BY dr.ThoiGian ASC;
 68  	 END IF;
 69  	 RETURN v_cursor;
 70  END;

SQL> -- 4. Tự động tính TongTien DONHANG khi thay đổi CHITIETDONHANG
SQL> CREATE OR REPLACE TRIGGER TRG_CTDH_TONGTIEN
  2  AFTER INSERT OR UPDATE OR DELETE ON CHITIETDONHANG
  3  FOR EACH ROW
  4  DECLARE
  5  	 v_Delta NUMBER := 0;
  6  	 v_MaDH VARCHAR2(10);
  7  	 v_LoaiKH NVARCHAR2(50);
  8  	 v_TyLeGiamGia NUMBER := 0;
  9  	 v_TongTienHangMoi NUMBER := 0;
 10  	 v_PhiVanChuyen NUMBER := 0;
 11  	 v_GiamGiaMoi NUMBER := 0;
 12  BEGIN
 13  	 -- 1. Xác định mức thay đổi của ThanhTien (Delta)
 14  	 IF INSERTING THEN
 15  	     v_Delta := :NEW.ThanhTien;
 16  	     v_MaDH := :NEW.MaDH;
 17  	 ELSIF UPDATING THEN
 18  	     v_Delta := :NEW.ThanhTien - :OLD.ThanhTien;
 19  	     v_MaDH := :NEW.MaDH;
 20  	 ELSIF DELETING THEN
 21  	     v_Delta := -:OLD.ThanhTien;
 22  	     v_MaDH := :OLD.MaDH;
 23  	 END IF;
 24  	 -- 2. Lấy Tổng tiền hàng hiện tại, Phí vận chuyển và Loại khách hàng
 25  	 SELECT DH.TongTienHang, DH.PhiVanChuyen, KH.LoaiKH
 26  	 INTO v_TongTienHangMoi, v_PhiVanChuyen, v_LoaiKH
 27  	 FROM DONHANG DH
 28  	 JOIN KHACHHANG KH ON DH.MaKH = KH.MaKH
 29  	 WHERE DH.MaDH = v_MaDH;
 30  	 v_TongTienHangMoi := NVL(v_TongTienHangMoi, 0) + v_Delta;
 31  	 -- 3. Xác định tỷ lệ giảm giá từ bảng THAMSO dựa trên LoaiKH
 32  	 BEGIN
 33  	     IF v_LoaiKH = 'Thường' THEN
 34  		 SELECT GiaTri INTO v_TyLeGiamGia FROM THAMSO WHERE TenTS = 'GG_THUONG';
 35  	     ELSIF v_LoaiKH = 'Thân thiết' THEN
 36  		 SELECT GiaTri INTO v_TyLeGiamGia FROM THAMSO WHERE TenTS = 'GG_THANTHIET';
 37  	     ELSIF v_LoaiKH = 'VIP' THEN
 38  		 SELECT GiaTri INTO v_TyLeGiamGia FROM THAMSO WHERE TenTS = 'GG_VIP';
 39  	     ELSE
 40  		 v_TyLeGiamGia := 0;
 41  	     END IF;
 42  	 EXCEPTION
 43  	     WHEN NO_DATA_FOUND THEN
 44  		 v_TyLeGiamGia := 0;
 45  	 END;
 46  	 -- 4. Tính toán số tiền Giảm giá mới
 47  	 v_GiamGiaMoi := v_TongTienHangMoi * v_TyLeGiamGia;
 48  	 -- 5. Cập nhật TongTien = TongTienHang + PhiVanChuyen - GiamGia
 49  	 UPDATE DONHANG
 50  	 SET TongTienHang = v_TongTienHangMoi,
 51  	     GiamGia = v_GiamGiaMoi,
 52  	     TongTien = v_TongTienHangMoi + NVL(v_PhiVanChuyen, 0) - v_GiamGiaMoi
 53  	 WHERE MaDH = v_MaDH;
 54  END;

SQL> CREATE OR REPLACE TRIGGER TRG_CANHBAO_NHAP_HANG
  2  AFTER INSERT OR UPDATE OF SLConLai ON TONKHO

  9  	 v_TenSP NVARCHAR2(100);

 12  	 v_NoiDung NVARCHAR2(500);
 13  BEGIN

 26  	     RETURN;
 27  	 END;

 30  	 SELECT NVL(SUM(SLConLai), 0) INTO v_TongTon

 40  	 END IF;

 50  	     COMMIT;
 51  	     RETURN;
 52  	 END IF;
 53  
 54  	 -- 6. Chống trùng: Chỉ chèn nếu chưa có thông báo CHƯA ĐỌC tương tự

SQL> CREATE OR REPLACE PROCEDURE SP_KIEMTRA_HETHAN_THONGBAO IS
  2  	 v_SoNgayHSD NUMBER;
  3  	 v_MinTonKho NUMBER;
  4  	 v_NgayHienTai DATE := TRUNC(SYSDATE);
  5  BEGIN
  6  	 -- 1. Lấy cấu hình
  7  	 BEGIN SELECT GiaTri INTO v_SoNgayHSD FROM THAMSO WHERE TenTS = 'CANHBAO_HETHAN'; EXCEPTION WHEN NO_DATA_FOUND THEN v_SoNgayHSD := 7; END;
  8  	 BEGIN SELECT GiaTri INTO v_MinTonKho FROM THAMSO WHERE TenTS = 'MIN_TONKHO'; EXCEPTION WHEN NO_DATA_FOUND THEN v_MinTonKho := 20; END;
  9  
 10  	 -- 2. Cập nhật HẾT HẠN
 11  	 UPDATE TONKHO SET TrangThai = N'Hết hạn', SLKhaDung = 0 WHERE TGHetHan < v_NgayHienTai AND TrangThai <> N'Hết hạn';
 12  	 -- 3. Cập nhật SẮP HẾT HẠN
 13  	 UPDATE TONKHO SET TrangThai = N'Sắp hết hạn' WHERE TGHetHan >= v_NgayHienTai AND (TGHetHan - v_NgayHienTai) <= v_SoNgayHSD AND TrangThai = N'Còn hạn';
 14  
 15  	 -- 4. Thông báo Hết hạn (Lô)
 16  	 FOR tk IN (
 17  	     SELECT TK.MaTonKho, SP.TenSP, TK.TGHetHan FROM TONKHO TK JOIN SANPHAM SP ON TK.MaSP = SP.MaSP WHERE TK.TrangThai = N'Hết hạn'
 18  	 ) LOOP
 19  	     DECLARE
 20  		 v_Msg NVARCHAR2(500) := N'Lô ' || tk.MaTonKho || N' của SP ' || tk.TenSP || N' đã HẾT HẠN vào ngày ' || TO_CHAR(tk.TGHetHan, 'DD/MM/YYYY');
 21  		 v_Count NUMBER;
 22  	     BEGIN
 23  		 SELECT COUNT(*) INTO v_Count FROM THONGBAO WHERE NoiDung = v_Msg;
 24  		 IF v_Count = 0 THEN INSERT INTO THONGBAO (MaTB, LoaiTB, NoiDung, TrangThaiTB, TGTao, NguoiNhan) VALUES ('TB' || LPAD(SEQ_THONGBAO.NEXTVAL, 8, '0'), N'Hết hạn', v_Msg, 0, SYSDATE, N'Nhân viên kho/thu mua'); END IF;
 25  	     END;
 26  	 END LOOP;
 27  
 28  	 -- 5. Thông báo Sắp hết hạn (Lô)
 29  	 FOR tk IN (
 30  	     SELECT TK.MaTonKho, SP.TenSP, TK.TGHetHan FROM TONKHO TK JOIN SANPHAM SP ON TK.MaSP = SP.MaSP WHERE TK.TrangThai = N'Sắp hết hạn'
 31  	 ) LOOP
 32  	     DECLARE
 33  		 v_Msg NVARCHAR2(500) := N'Lô ' || tk.MaTonKho || N' của SP ' || tk.TenSP || N' sẽ hết hạn vào ' || TO_CHAR(tk.TGHetHan, 'DD/MM/YYYY');
 34  		 v_Count NUMBER;
 35  	     BEGIN
 36  		 SELECT COUNT(*) INTO v_Count FROM THONGBAO WHERE NoiDung = v_Msg;
 37  		 IF v_Count = 0 THEN INSERT INTO THONGBAO (MaTB, LoaiTB, NoiDung, TrangThaiTB, TGTao, NguoiNhan) VALUES ('TB' || LPAD(SEQ_THONGBAO.NEXTVAL, 8, '0'), N'Sắp hết hạn', v_Msg, 0, SYSDATE, N'Nhân viên kho/thu mua'); END IF;
 38  	     END;
 39  	 END LOOP;
 40  
 41  	 -- 6. TÍNH TỔNG TỒN KHO CHO MỖI SẢN PHẨM
 42  	 -- (Đã xóa bỏ logic ép hàng hết hạn về 0. Giờ hệ thống tính tổng theo số lượng vật lý thực tế)
 43  	 FOR rec IN (
 44  	     SELECT SP.MaSP, SP.TenSP,
 45  		    NVL(SUM(TK.SLConLai), 0) AS TongTon
 46  	     FROM SANPHAM SP
 47  	     LEFT JOIN TONKHO TK ON SP.MaSP = TK.MaSP
 48  	     GROUP BY SP.MaSP, SP.TenSP
 49  	 ) LOOP
 50  	     DECLARE
 51  		 v_Msg NVARCHAR2(500);
 52  		 v_Type NVARCHAR2(50);
 53  		 v_Count NUMBER;
 54  	     BEGIN
 55  		 IF rec.TongTon <= 0 THEN
 56  		     v_Type := N'Hết hàng';
 57  		     v_Msg := N'CẢNH BÁO: Sản phẩm [' || rec.TenSP || N'] đã HẾT HÀNG hoàn toàn. Yêu cầu nhập hàng mới ngay!';
 58  		 ELSIF rec.TongTon < v_MinTonKho THEN
 59  		     v_Type := N'Sắp hết hàng';
 60  		     v_Msg := N'Thông báo: Sản phẩm [' || rec.TenSP || N'] sắp hết hàng. Hiện chỉ còn ' || rec.TongTon || N' đơn vị. Yêu cầu nhập thêm hàng!';
 61  		 ELSE
 62  		     CONTINUE;
 63  		 END IF;
 64  
 65  		 -- Logic chống trùng lặp theo ngày
 66  		 SELECT COUNT(*) INTO v_Count
 67  		 FROM THONGBAO
 68  		 WHERE LoaiTB = v_Type
 69  		   AND NoiDung LIKE N'%[' || rec.TenSP || N']%'
 70  		   AND TRUNC(TGTao) = TRUNC(SYSDATE);
 71  
 72  		 IF v_Count = 0 THEN
 73  		     INSERT INTO THONGBAO (MaTB, LoaiTB, NoiDung, TrangThaiTB, TGTao, NguoiNhan)
 74  		     VALUES ('TB' || LPAD(SEQ_THONGBAO.NEXTVAL, 8, '0'), v_Type, v_Msg, 0, SYSDATE, N'Nhân viên kho/thu mua');
 75  		 END IF;
 76  	     END;
 77  	 END LOOP;
 78  
 79  	 COMMIT;
